const H='0x'+'ab'.repeat(32)
const A='0x20C0000000000000000000000000000000000000'
const origins=[['explore.tempo.xyz','explore.tempo.xyz','Mainnet'],['explore.4217.tempo.xyz','explore.tempo.xyz','Mainnet'],['explore.mainnet.tempo.xyz','explore.tempo.xyz','Mainnet'],['explore.presto.tempo.xyz','explore.tempo.xyz','Mainnet'],['explore.42431.tempo.xyz','explore.testnet.tempo.xyz','Testnet'],['explore.moderato.tempo.xyz','explore.testnet.tempo.xyz','Testnet'],['explore.testnet.tempo.xyz','explore.testnet.tempo.xyz','Testnet'],['explore.31318.tempo.xyz','explore.devnet.tempo.xyz','Devnet'],['explore.devnet.tempo.xyz','explore.devnet.tempo.xyz','Devnet'],['explore.nextfork.devnet.tempo.xyz','explore.nextfork.devnet.tempo.xyz','Nextfork'],['explore.zone-prover.devnet.tempo.xyz','explore.zone-prover.devnet.tempo.xyz','Zone Prover']]
